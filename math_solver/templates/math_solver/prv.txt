        // AI Output functionality
        const outputText = document.getElementById('output-text');

        async function fetchAIOutput() {
            try {
                const response = await fetch('/get-ai-output/');
                const data = await response.json();
                console.log('AI Output:', data.output); // Log AI output
                if (data.output) {
                    outputText.innerText = data.output; // Update AI response on the webpage
                }
            } catch (error) {
                console.error('Error fetching AI output:', error);
            }
        }

        // Fetch AI output every 5 seconds
        setInterval(fetchAIOutput, 5000);

        // Add loading animation when waiting for AI response
        function showLoading() {
